\#{ resumeow — "swiss": minimalist single-column CV in the International Typographic Style. Section labels live in a fixed left gutter (placed with \llap so body text still flows and page-breaks naturally); one small photo; grayscale only — the YAML `color` is used solely as a near-black ink for labels and hairlines. Same data + same render.py as default.tex. }
% !TeX program = pdflatex
\documentclass[11pt,a4paper]{article}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[british]{babel}
\usepackage[left=22mm, right=20mm, top=18mm, bottom=18mm]{geometry}
\usepackage[stretch=25, shrink=25, tracking=true]{microtype}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{enumitem}

\usepackage{FiraSans}
\renewcommand{\familydefault}{\sfdefault}

% --- Palette: grayscale only. `ink` comes from the YAML `color` (a near-black),
% everything else is a tint of gray. No accent hue -> deliberate minimality. ----
\definecolor{ink}{HTML}{\VAR{ color }}
\definecolor{mute}{gray}{0.45}   % secondary text: dates, meta labels
\definecolor{hair}{gray}{0.80}   % hairline rules
\definecolor{chip}{gray}{0.94}   % tool-chip background

% Left gutter geometry: section labels sit in `\labelw`, body starts at `\gutter`.
\newlength{\gutter}\setlength{\gutter}{34mm}
\newlength{\labelw}\setlength{\labelw}{27mm}

% Inline "tool chip" — minimal monospace token, faint gray plate (see YAML \tool{}).
\newcommand{\tool}[1]{{\setlength{\fboxsep}{2pt}\colorbox{chip}{\footnotesize\ttfamily #1}}}

% Section label dropped into the left gutter; body keeps flowing beside/under it.
\newcommand{\sect}[1]{%
  \par\addvspace{2.6ex}%
  \noindent\llap{\smash{\parbox[t]{\labelw}{\raggedright\footnotesize\bfseries\scshape\color{ink}\textls[40]{#1}}}\hspace{\dimexpr\gutter-\labelw\relax}}%
  \ignorespaces}
\newcommand{\dates}[1]{\hfill{\small\color{mute}#1}}

\usepackage[hidelinks]{hyperref}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5ex}
\setlist{nosep, leftmargin=1.3em, label={\textcolor{hair}{\textbf{\textendash}}}}

\begin{document}
\pagestyle{empty}

\begin{minipage}[b]{28mm}

\BLOCK{ endif }

\BLOCK{ if transversal_competences }
