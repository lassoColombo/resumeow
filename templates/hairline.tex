\#{ resumeow — "hairline": a light two-column CV. Unlike default/sleek (which fill the sidebar with a solid colour block), here a narrow left column is separated from the main column by a single hairline rule — no fills, grayscale only. Uses `paracol` so both columns break across pages naturally. The YAML `color` is used purely as a near-black ink. Same data + same render.py as default.tex. }
% !TeX program = pdflatex
\documentclass[10pt,a4paper]{article}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[british]{babel}
\usepackage[left=16mm, right=16mm, top=16mm, bottom=16mm]{geometry}
\usepackage[stretch=25, shrink=25, tracking=true]{microtype}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{marvosym}
\usepackage{enumitem}
\usepackage{paracol}

\usepackage{FiraSans}
\renewcommand{\familydefault}{\sfdefault}

% --- Grayscale palette. `ink` = YAML `color` (near-black); rest are gray tints. --
\definecolor{ink}{HTML}{\VAR{ color }}
\definecolor{mute}{gray}{0.45}
\definecolor{hair}{gray}{0.75}
\definecolor{chip}{gray}{0.94}

\newcommand{\tool}[1]{{\setlength{\fboxsep}{2pt}\colorbox{chip}{\footnotesize\ttfamily #1}}}
\newcommand{\dates}[1]{\hfill{\small\color{mute}#1}}

% Sidebar section label: small caps, tiny, with a faint rule under it.
\newcommand{\headleft}[1]{\vspace*{2.2ex}{\scriptsize\bfseries\scshape\color{ink}\textls[60]{#1}}\par%
  \vspace*{0.4ex}{\color{hair}\hrulefill}\par\vspace*{0.9ex}}
% Main-column section label: larger small caps, short ink rule beneath.
\newcommand{\headright}[1]{\vspace*{2.6ex}{\large\bfseries\scshape\color{ink}\textls[40]{#1}}\par%
  \vspace*{0.5ex}{\color{ink}\rule{2.6em}{1pt}}\par\vspace*{1.1ex}}
\newcommand{\competence}[2]{{\bfseries\color{ink}#1}\par\vspace{0.25em}#2\par\vspace{0.7em}}

\usepackage[hidelinks]{hyperref}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\setlist{nosep, leftmargin=1.2em, label={\textcolor{hair}{\textbf{\textendash}}}}

% Two columns: narrow left (0.30), wide right; a hairline between them.
\columnratio{0.30}
\setlength{\columnsep}{7mm}
\setlength{\columnseprule}{0.4pt}
\colorlet{col-sep}{hair}

\begin{document}
\pagestyle{empty}

\begin{paracol}{2}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% LEFT COLUMN %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\raggedright
\null\hfill\includegraphics[width=0.72\linewidth]{\VAR{ img }}\hfill\null
\vspace{1.6ex}
{\centering\bfseries\large\textls[20]{\VAR{ name }}\par}

\headleft{\VAR{ i18n.contacts }}
{\footnotesize
\MVAt\ \VAR{ contacts.mail }\\[0.5ex]
\Mobilefone\ \VAR{ contacts.phone }%
\BLOCK{ for link in contacts.links or [] }
\\[0.5ex]\Mundus\ \href{\VAR{ link.link }}{\VAR{ link.display }}%
\BLOCK{ endfor }
}

\headleft{\VAR{ i18n.languages }}
\begin{itemize}
\BLOCK{ for lang in languages }
\item {\footnotesize \VAR{ lang }}
\BLOCK{ endfor }
\end{itemize}

\headleft{\VAR{ i18n.education }}
{\footnotesize
\BLOCK{ for ed in education }
{\bfseries\color{ink}\VAR{ ed.title }}\\
\textcolor{mute}{\VAR{ ed.institution }}\\
\textcolor{mute}{\VAR{ i18n.grade_label } \VAR{ ed.grade }}\par\vspace{0.8ex}
\BLOCK{ endfor }
}

\switchcolumn
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% RIGHT COLUMN %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\headright{\VAR{ i18n.profile }}
\BLOCK{ if profile is string }
\VAR{ profile }
\BLOCK{ else }
\begin{itemize}
\BLOCK{ for item in profile }
\item \VAR{ item }
\BLOCK{ endfor }
\end{itemize}
\BLOCK{ endif }

\headright{\VAR{ i18n.desiderata }}
\BLOCK{ if desiderata is string }
\VAR{ desiderata }
\BLOCK{ else }
\begin{itemize}
\BLOCK{ for item in desiderata }
\item \VAR{ item }
\BLOCK{ endfor }
\end{itemize}
\BLOCK{ endif }

\headright{\VAR{ i18n.key_competences }}
\BLOCK{ for c in key_competences }
\competence{\VAR{ c.name }}{\VAR{ c.text }}
\BLOCK{ endfor }

\BLOCK{ if transversal_competences }
\headright{\VAR{ i18n.transversal_competences }}
\BLOCK{ for c in transversal_competences }
\competence{\VAR{ c.name }}{\VAR{ c.text }}
\BLOCK{ endfor }
\BLOCK{ endif }

\headright{\VAR{ i18n.work_experiences }}
\BLOCK{ for e in work_experiences }
{\bfseries\color{ink}\VAR{ e.title }}\,\textcolor{mute}{— \VAR{ e.company }}\dates{\VAR{ e.dates }}\par\vspace{0.25em}
\VAR{ e.description }
\BLOCK{ if 'items' in e }
\begin{itemize}
\BLOCK{ for it in e['items'] }
\item \textbf{\VAR{ it.project }:} \VAR{ it.text }
\BLOCK{ endfor }
\end{itemize}
\BLOCK{ endif }
\par\vspace{0.5em}{\color{hair}\hrulefill}\par\vspace{0.5em}
\BLOCK{ endfor }

\end{paracol}
\end{document}
